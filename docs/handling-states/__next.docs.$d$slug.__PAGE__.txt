import { quarter } from "./quarter";

export async function refund(intent: ActionIntent): Promise<ActionView> {
  const action = await send(intent);

  switch (action.state) {
    case "CONFIRMED":
      await markRefunded(intent.subject, action.output?.providerObjectId);
      return action;

    case "NOT_EXECUTED":
      await markRefundFailed(intent.subject, action.reason);
      return action;

    case "SAFE_TO_RETRY":
      // Same id, same input: Quarter reuses the idempotency key.
      return send(intent);

    case "NEEDS_REVIEW":
      await openReviewTicket(action.actionId, action.reason);
      return action;

    default:
      // AUTHORIZED, EXECUTING, OUTCOME_UNKNOWN, RECONCILING: the webhook finishes the job.
      await markRefundPending(intent.subject, action.actionId);
      return action;
  }
}

async function send(intent: ActionIntent): Promise<ActionView> {
  try {
    return await quarter.actions.execute(intent);
  } catch (error) {
    if (error instanceof QuarterTransportError && error.maybeExecuted) {
      return quarter.actions.execute(intent);
    }
    throw error;
  }
